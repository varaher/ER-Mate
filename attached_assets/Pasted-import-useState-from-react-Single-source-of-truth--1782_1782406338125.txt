// ── Billing toggle ─────────────────────────────────────────────────────────
function BillingToggle({ billing, onChange }) {
  return (
    <div style={{
      display: "flex", background: "#EEF0F2",
      borderRadius: 14, padding: 4, gap: 3,
    }}>
      {["monthly", "annual"].map(b => (
        <button key={b} onClick={() => onChange(b)} style={{
          flex: 1, border: "none", cursor: "pointer",
          background: billing === b ? "white" : "transparent",
          borderRadius: 11, padding: "10px 16px",
          fontSize: 13, fontWeight: 700,
          color: billing === b ? "#0D1117" : "#9CA3AF",
          boxShadow: billing === b ? "0 2px 8px rgba(0,0,0,0.08)" : "none",
          transition: "all 0.2s", fontFamily: "inherit",
          display: "flex", alignItems: "center", justifyContent: "center", gap: 6,
        }}>
          {b === "monthly" ? "Monthly" : "Annual"}
          {b === "annual" && (
            <span style={{
              fontSize: 9, fontWeight: 800,
              background: billing === "annual" ? "rgba(30,184,112,0.15)" : "rgba(156,163,175,0.15)",
              color: billing === "annual" ? "#15924F" : "#9CA3AF",
              borderRadius: 5, padding: "2px 6px",
            }}>SAVE 17%</span>
          )}
        </button>
      ))}
    </div>
  );
}
