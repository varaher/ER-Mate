// ── Access control — single source of truth ────────────────────────────────
function getAccess(sub) {
  const unlimited = sub.teamActive || sub.plan === "pro";
  return {
    canDocument:        sub.plan !== null,
    unlimitedCases:     unlimited,
    smartDictation:     unlimited,
    dischargeSummary:   unlimited,
    decisionSupport:    unlimited,
    documentScanning:   unlimited,
    abgInterpretation:  unlimited,
    emReference:        unlimited,
    rounds:             unlimited,
    clinicalMemory:     sub.plan === "pro", // personal layer — pro only
    handoverSystem:     sub.teamActive,
    shiftManagement:    sub.teamActive,
    hodDashboard:       sub.teamActive && sub.teamRole === "hod",
    showCredits:        false, // removed from new plans
    showUpgrade:        !unlimited,
    showAddPro:         sub.teamActive && sub.plan !== "pro",
  };
}

function TeamEnrollButton({ consultants, residents, billing, sticky = false }) {
  const [loading, setLoading] = useState(false);
  const [step, setStep] = useState("idle"); // idle | confirming | processing | success

  const monthlyAmount = (consultants * 59900) + (residents * 39900); // paise
  const annualAmount  = Math.round(monthlyAmount * 10);
  const displayAmount = billing === "annual" ? annualAmount : monthlyAmount;
  const totalDoctors  = consultants + residents;

  const handleEnroll = async () => {
    if (totalDoctors < 4) {
      alert("Minimum 4 doctors required to activate a Team plan.\nPlease add at least 4 doctors using the calculator above.");
      return;
    }
    setStep("confirming");
  };

  const handleConfirm = async () => {
    setLoading(true);
    setStep("processing");

    try {
      // 1. Call backend to create Razorpay subscription
      // const res = await fetch('/api/subscription/team/create', {
      //   method: 'POST',
      //   headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
      //   body: JSON.stringify({ consultants, residents, billingCycle: billing }),
      // });
      // const { subscriptionId, keyId } = await res.json();

      // Demo values:
      const subscriptionId = "sub_team_demo_001";
      const keyId = "rzp_test_XXXXXXXXXX";

      // 2. Open Razorpay checkout
      openRazorpayCheckout({
        subscriptionId,
        keyId,
        amount: displayAmount,
        name: "Dr. Neeraj",
        email: "neeraj@ermate.in",
        phone: "+919999999999",
        onSuccess: (data) => {
          setStep("success");
          setLoading(false);
          // Webhook handles activation — UI just shows confirmation
        },
        onDismiss: () => {
          setStep("idle");
          setLoading(false);
        },
      });
    } catch (err) {
      console.error("Team enrollment error:", err);
      setStep("idle");
      setLoading(false);
    }
  };

  // Success state
  if (step === "success") {
    return (
      <div style={{ background:C.greenLight, border:`1px solid ${C.greenBorder}`,
        borderRadius:14, padding:"16px", textAlign:"center" }}>
        <div style={{ fontSize:24, marginBottom:8 }}>✅</div>

          Team plan activated!

          They'll receive an invite and can join immediately.
        </div>
      </div>
    );
  }

  // Confirmation modal
  if (step === "confirming") {
    return (
      <div style={{ background:"rgba(255,255,255,0.97)", borderRadius:16,

            fontSize:13, fontWeight:700, color:"white", cursor:"pointer",

          </>
        ) : (
          <>
            <Ic.People c="white" size={16}/>
            Enroll Now

              <span style={{
                background:"rgba(255,255,255,0.2)", borderRadius:6,
