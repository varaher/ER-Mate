```typescript
// server/services/sarvam.ts or 
// wherever transcript is processed
// after getting the raw transcript back:

function cleanTranscript(raw: string): string {
  // Split into sentences
  const sentences = raw.split(/[.!?]+/).map(s => s.trim()).filter(Boolean);
  
  // Detect repeated phrases (loop bug)
  const seen = new Map<string, number>();
  const cleaned: string[] = [];
  
  for (const sentence of sentences) {
    const normalized = sentence.toLowerCase().trim();
    const count = seen.get(normalized) || 0;
    
    if (count < 2) { // allow max 2 occurrences
      cleaned.push(sentence);
      seen.set(normalized, count + 1);
    }
    // else: skip — it's a loop repetition
  }
  
  // If more than 60% of sentences are duplicates
  // the whole recording is noise — reject it
  const uniqueRatio = cleaned.length / sentences.length;
  if (uniqueRatio < 0.4 && sentences.length > 10) {
    throw new Error('TRANSCRIPT_LOOP_DETECTED');
  }
  
  return cleaned.join('. ');
}

// In the route, catch this error:
try {
  const cleanedTranscript = cleanTranscript(rawTranscript);
  // proceed with GPT-4o extraction
} catch (err) {
  if (err.message === 'TRANSCRIPT_LOOP_DETECTED') {
    return res.status(400).json({
      error: 'Background noise detected. Please record again in a quieter environment.',
      code: 'NOISE_DETECTED'
    });
  }
}
```

```typescript
function renderCaseNote(data: CaseData) {
  const v = data.vitals || {};
  const h = data.history || {};
  const p = data.primary || {};
  const e = data.examination || {};
  const t = data.treatment || {};
  const d = data.disposition || {};

  // Helper — show value or fallback
  const val = (v: any, fallback = 'Not mentioned') => 
    v && v !== '' ? v : fallback;
  
  // Helper — highlight abnormal vitals
  const vitalVal = (v: any, field: string) => {
    if (!v || v === '') return '—';
    const abnormal = isAbnormal(v, field);
    return { value: v, abnormal };
  };

  return `
PATIENT
  Name        ${val(data.patientName, 'Not recorded')}
  Age/Sex     ${val(data.age, '—')}${data.gender ? ' · ' + data.gender : ''}
  Priority    ${data.priority || 'P4'}
  Complaint   ${val(h.symptoms || data.chiefComplaint)}

VITALS
  HR          ${vitalVal(v.hr, 'hr')}
  BP          ${vitalVal(v.bp, 'bp')}
  SpO₂        ${vitalVal(v.spo2, 'spo2')}
  RR          ${vitalVal(v.rr, 'rr')}
  Temp        ${vitalVal(v.temp, 'temp')}
  GCS         ${vitalVal(v.gcs, 'gcs')}
  GRBS        ${vitalVal(v.grbs, 'grbs')}

HISTORY (SAMPLE)
  Symptoms    ${val(h.symptoms)}
  Allergies   ${val(h.allergies, 'NKDA')}
  Medications ${val(h.medications, 'None')}
  Past Hx     ${val(h.pastHistory, 'Nil significant')}
  Last Meal   ${val(h.lastMeal)}
  Events      ${val(h.events)}

PRIMARY SURVEY (ABCDE)
  Airway      ${val(p.airway, 'Patent')}
  Breathing   ${val(p.breathing)}
  Circulation ${val(p.circulation)}
  Disability  ${val(p.disability)}
  Exposure    ${val(p.exposure)}

EXAMINATION
  General     ${val(e.general)}
  CVS         ${val(e.cvs)}
  Respiratory ${val(e.respiratory)}
  Abdomen     ${val(e.abdomen)}
  Neuro       ${val(e.neuro)}

TREATMENT GIVEN
  Medications ${formatMeds(t.medications)}
  Procedures  ${val(t.procedures, 'None')}
  IV Access   ${val(t.ivAccess)}
  Fluids      ${val(t.fluids, 'None')}
  Oxygen      ${val(t.oxygen, 'Not required')}

IMPRESSION
  Diagnosis   ${val(d.diagnosis)}
  
PLAN
  ${val(d.plan)}
  Decision    ${val(d.decision, 'Pending')}
  `;
}

function isAbnormal(value: string, field: string): boolean {
  const num = parseFloat(value);
  if (isNaN(num)) return false;
  
  const ranges: Record<string, [number, number]> = {
    hr:   [60, 100],
    spo2: [95, 100],
    rr:   [12, 20],
    temp: [36.1, 37.8],
    gcs:  [15, 15],
    grbs: [70, 140],
  };
  
  const range = ranges[field];
  if (!range) return false;
  return num < range[0] || num > range[1];
}

function formatMeds(meds: any): string {
  if (!meds) return 'None given';
  if (typeof meds === 'string') return meds;
  if (Array.isArray(meds)) {
    return meds.map(m => 
      `${m.drug} ${m.dose} ${m.route}${m.frequency ? ' ' + m.frequency : ''}`
    ).join('\n              ');
  }
  return String(meds);
}
```
