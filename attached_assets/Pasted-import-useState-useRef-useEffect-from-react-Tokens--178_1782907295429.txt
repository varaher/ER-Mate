// ── Document card component ────────────────────────────────────────────────
function DocCard({ type, title, icon, children, onCopy, onEdit, onExport, onSave }) {
  const [copied, setCopied] = useState(false);

  const handleCopy = () => {
    setCopied(true);
    onCopy?.();
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div style={{
      background: C.docBg,
      borderRadius: 16,
      border: `1px solid ${C.docBorder}`,
      overflow: "hidden",
      boxShadow: "0 4px 20px rgba(0,0,0,0.18), 0 1px 4px rgba(0,0,0,0.1)",
      marginBottom: 4,
    }}>
      {/* Document header */}
      <div style={{
        background: type === "discharge"
          ? "linear-gradient(135deg,#1E2530,#252E3D)"
          : "linear-gradient(135deg,#0B0F14,#1A2332)",
        padding: "11px 14px",
        display: "flex", alignItems: "center", gap: 8,
        borderBottom: `1px solid rgba(255,255,255,0.06)`,
      }}>
        <span style={{ fontSize: 14 }}>{icon}</span>
        <span style={{ fontSize: 12, fontWeight: 800, color: "rgba(255,255,255,0.9)",
          letterSpacing: 0.3, flex: 1 }}>{title}</span>
        <span style={{ fontSize: 9, fontWeight: 700, color: C.green,
          background: C.greenLight, borderRadius: 4, padding: "2px 7px",
          border: `1px solid ${C.greenBd}` }}>
          {type === "discharge" ? "DISCHARGE" : "CASE NOTE"}
        </span>
      </div>

      {/* Document body */}
      <div style={{ padding: "14px 16px", fontFamily: "'Georgia',serif" }}>
        {children}
      </div>

      {/* Action bar */}
      <div style={{
        borderTop: `1px solid ${C.docBorder}`,
        padding: "9px 14px",
        background: C.surface,
        display: "flex", gap: 6,
      }}>
        <DocAction icon={<CopyIcon/>} label={copied ? "Copied ✓" : "Copy all"}
          onClick={handleCopy} primary={!copied} done={copied}/>
        {onEdit && <DocAction icon={<EditIcon/>} label="Edit" onClick={onEdit}/>}
        {onExport && <DocAction icon={<PdfIcon/>} label="Export PDF" onClick={onExport}/>}
        {onSave && <DocAction icon={<SaveIcon/>} label="Save" onClick={onSave}/>}
      </div>
    </div>
  );
}

function DocAction({ icon, label, onClick, primary, done }) {
  return (
    <button onClick={onClick} style={{
      display: "flex", alignItems: "center", gap: 5,
      background: done ? C.greenLight : primary ? C.ink : "white",
      border: `1px solid ${done ? C.greenBd : primary ? "transparent" : C.border}`,
      borderRadius: 8, padding: "6px 11px",
      fontSize: 11, fontWeight: 700,
      color: done ? C.green : primary ? "white" : C.muted,
      cursor: "pointer", fontFamily: "inherit", transition: "all 0.15s",
    }}>{icon}{label}</button>
  );
}

                <SystemLabel text="ErMate processed your dictation"/>

                <ErMateResponse>
                  <div className="msg-in" style={{ marginBottom: 4 }}>
                    <div style={{ fontSize: 12, color: "rgba(255,255,255,0.5)",
                      marginBottom: 10, lineHeight: 1.5 }}>
                      Case note ready. 6 of 7 sections captured.
                      Vitals, history, and treatment filled automatically.
                    </div>
                    <CaseNoteDoc
                      onCopy={() => showToast("Case note copied to clipboard")}
                      onEdit={() => {}}
                      onExport={() => showToast("Exporting PDF...")}
                      onSave={() => showToast("Saved to dashboard")}
                    />
                  </div>
                </ErMateResponse>

                {step === 1 && suggestions.length > 0 && (
                  <div className="msg-in">
                    <SuggestionChips onTap={handleChip} suggestions={suggestions}/>
                  </div>
                )}
              </>
            )}

            {/* Discharge summary */}
            {step >= 2 && (
              <>
                <DoctorBubble text="Prepare discharge summary"/>
                <ErMateResponse>
                  <div className="msg-in" style={{ marginBottom: 4 }}>
                    <div style={{ fontSize: 12, color: "rgba(255,255,255,0.5)",
                      marginBottom: 10 }}>
                      Discharge summary generated from your case note.
                    </div>
                    <DischargeSummaryDoc
                      onCopy={() => showToast("Discharge summary copied")}
                      onExport={() => showToast("Exporting PDF...")}
                    />
                  </div>
                </ErMateResponse>
                {step === 2 && (
                  <div className="msg-in">
                    <SuggestionChips onTap={handleChip} suggestions={suggestions}/>
                  </div>
                )}
              </>
            )}

            {/* Update confirmation */}
            {step >= 3 && (
              <>
                <DoctorBubble text="Add allergy — penicillin"/>
                <ErMateResponse>
                  <div className="msg-in">
                    <UpdateConfirmation
                      text="Allergy updated: Penicillin added. Case note and discharge summary updated automatically."/>
                    <div style={{ fontSize: 11, color: "rgba(255,255,255,0.3)",
                      marginTop: 8 }}>
                      Both documents reflect this change. Tap Copy again to get the latest version.
                    </div>
                  </div>
                </ErMateResponse>
                {step === 3 && (
                  <div className="msg-in">
                    <SuggestionChips onTap={handleChip} suggestions={suggestions}/>
                  </div>
                )}
              </>
            )}

            {/* Referral */}
            {step >= 4 && (
              <>
                <DoctorBubble text="Referral letter to cardiology"/>
                <ErMateResponse>
                  <div className="msg-in">
                    <DocCard type="referral" title="Referral Letter · Cardiology" icon="📨"
                      onCopy={() => showToast("Referral letter copied")}
                      onExport={() => showToast("Exporting PDF...")}>
                      <div style={{ fontSize: 11.5, color: C.ink, lineHeight: 1.8,
                        fontFamily: "'Inter',sans-serif" }}>
                        <div style={{ marginBottom: 10, color: C.muted, fontSize: 10.5 }}>
                          01 July 2026 · City Emergency Department
                        </div>
                        <p style={{ marginBottom: 10 }}>
                          <strong>To:</strong> Consultant Cardiologist
                        </p>
                        <p style={{ marginBottom: 10 }}>
                          We are referring <strong>Mr. Abdulahad, 52M</strong> with
                          acute anterior STEMI (ST elevation V1–V4). Presented at 14:32
                          with crushing chest pain × 2 hours. Background T2DM and hypertension.
                        </p>
                        <p style={{ marginBottom: 10 }}>
                          <strong>Treatment given:</strong> Aspirin 325mg, Morphine 5mg IV,
                          Heparin 5000 IU IV. Haemodynamically compromised on arrival.
                          Cath lab activation requested.
                        </p>
                        <p>Kindly review and manage accordingly. Patient is on continuous
                          cardiac monitoring.</p>
                      </div>
                    </DocCard>
                  </div>
                </ErMateResponse>
              </>
            )}

          </div>

          {/* Suggestions above input */}
          {step >= 1 && step >= 4 && (
            <div style={{ padding: "0 14px", flexShrink: 0 }}>
              <SuggestionChips
                onTap={handleChip}
                suggestions={["Export all as PDF", "Save & close", "Add to notes"]}
              />
            </div>
          )}

          {/* Input bar */}
          <div style={{
            background: "rgba(255,255,255,0.03)",
            borderTop: "1px solid rgba(255,255,255,0.07)",
            padding: "10px 14px 14px", flexShrink: 0,
          }}>
            <div style={{ display: "flex", gap: 10, alignItems: "flex-end" }}>

              {/* Mic button */}
              <button onClick={handleMic} style={{
                width: 42, height: 42, borderRadius: 12, border: "none",
                background: step === 0
                  ? `linear-gradient(135deg,${C.green},${C.greenDark})`
                  : "rgba(29,184,112,0.1)",
                cursor: "pointer", display: "flex",
                alignItems: "center", justifyContent: "center",
                flexShrink: 0,
                boxShadow: step === 0 ? "0 4px 14px rgba(29,184,112,0.3)" : "none",
              }}>
                <MicIcon recording={recording}/>
              </button>

              {/* Text input */}
              <div style={{ flex: 1, position: "relative" }}>
                <input
                  value={input}
                  onChange={e => setInput(e.target.value)}
                  onKeyDown={e => e.key === "Enter" && handleSend()}
                  placeholder={step === 0
                    ? "Or type your case note..."
                    : "Try: discharge summary, add allergy, referral letter..."}
                  style={{
                    width: "100%", background: "rgba(255,255,255,0.06)",
                    border: "1px solid rgba(255,255,255,0.1)",
                    borderRadius: 12, padding: "11px 14px",
                    fontSize: 13, color: "rgba(255,255,255,0.85)",
                    fontFamily: "inherit", outline: "none",
                  }}
                />
              </div>

              {/* Send button */}
              <button onClick={handleSend} style={{
                width: 42, height: 42, borderRadius: 12, border: "none",
                background: input
                  ? `linear-gradient(135deg,${C.green},${C.greenDark})`
                  : "rgba(255,255,255,0.06)",
                cursor: input ? "pointer" : "default",
                display: "flex", alignItems: "center", justifyContent: "center",
                flexShrink: 0, fontSize: 16,
                boxShadow: input ? "0 4px 14px rgba(29,184,112,0.3)" : "none",
              }}>
                <span style={{ color: input ? "white" : "rgba(255,255,255,0.25)" }}>↑</span>
              </button>
            </div>
          </div>

          {/* Toast */}
          {toast && (
            <div style={{
              position: "absolute", top: 70, left: "50%",
              transform: "translateX(-50%)",
              background: C.greenDark, color: "white",
              fontSize: 12, fontWeight: 700,
              padding: "8px 18px", borderRadius: 99,
              boxShadow: "0 4px 16px rgba(0,0,0,0.3)",
              whiteSpace: "nowrap", zIndex: 100,
              animation: "slideUp 0.25s ease",
            }}>✓ {toast}</div>
          )}

        </div>
      </div>

      {/* Flow guide */}
      <div style={{
        background: "white", borderRadius: 16, padding: "16px 20px",
        maxWidth: 390, width: "100%", border: "1px solid #E8EAED",
      }}>
        <div style={{ fontSize: 11, fontWeight: 800, color: "#9CA3AF",
          letterSpacing: 1, textTransform: "uppercase", marginBottom: 10 }}>
          Try the flow
        </div>
        {[
          "Tap 🎙 mic — watch dictation processing",
          "Case note appears as a clinical document",
          "Tap 'Prepare discharge summary' chip",
          "Discharge summary appears below",
          "Tap 'Add allergy — penicillin'",
          "Both documents update automatically",
          "Tap 'Referral letter to cardiology'",
          "Copy any document with one tap",
        ].map((s,i) => (
          <div key={i} style={{ display:"flex", gap:8, marginBottom: i<7?6:0 }}>
            <span style={{ color: C.green, fontWeight: 700, flexShrink: 0 }}>{i+1}</span>
            <span style={{ fontSize:12, color:"#6B7280", lineHeight:1.5 }}>{s}</span>
          </div>
        ))}
      </div>

    </div>
  );
}
