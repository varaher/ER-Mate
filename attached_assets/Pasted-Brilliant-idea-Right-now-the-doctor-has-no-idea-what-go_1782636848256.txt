## What to Tell Your Developer

**Backend — no changes needed.** GPT-4o already extracts and fills fields. The completion map is purely frontend.

**Frontend changes:**

```typescript
// 1. After Smart Dictation completes,
//    calculate completion per tab

function calculateCompletion(formData) {
  return {
    patient: {
      filled: countFilled([
        formData.name, formData.age, 
        formData.gender, formData.chiefComplaint,
        formData.vitals?.hr, formData.vitals?.bp,
        formData.vitals?.spo2, formData.vitals?.rr,
        formData.vitals?.temp, formData.vitals?.gcs,
        formData.vitals?.grbs,
      ]),
      total: 11,
    },
    history: {
      filled: countFilled([
        formData.sample?.signsSymptoms,
        formData.sample?.allergies,
        formData.sample?.medications,
        formData.sample?.pastHistory,
        formData.sample?.lastMeal,
        formData.sample?.events,
        formData.sample?.familyHistory,
        formData.sample?.socialHistory,
      ]),
      total: 8,
    },
    primary: {
      filled: countFilled([
        formData.abcde?.airway,
        formData.abcde?.breathing,
        formData.abcde?.circulation,
        formData.abcde?.disability,
        formData.abcde?.exposure,
        formData.abcde?.vbg,
        formData.abcde?.ecg,
      ]),
      total: 7,
    },
    examination: {
      filled: countFilled([
        formData.exam?.general,
        formData.exam?.headNeck,
        formData.exam?.chest,
        formData.exam?.cvs,
        formData.exam?.abdomen,
        formData.exam?.neuro,
        formData.exam?.extremities,
        formData.exam?.skin,
      ]),
      total: 8,
    },
    treatment: {
      filled: countFilled([
        formData.treatment?.medications,
        formData.treatment?.ivAccess,
        formData.treatment?.ivFluids,
        formData.treatment?.procedures,
        formData.treatment?.oxygen,
        formData.treatment?.consults,
      ]),
      total: 6,
    },
    notes: {
      filled: countFilled([formData.addendumNotes]),
      total: 1,
    },
    disposition: {
      filled: countFilled([
        formData.disposition?.diagnosis,
        formData.disposition?.plan,
        formData.disposition?.followUp,
        formData.disposition?.instructions,
      ]),
      total: 4,
    },
  };
}

function countFilled(fields) {
  return fields.filter(f => 
    f !== null && 
    f !== undefined && 
    f !== "" && 
    f !== "Not mentioned"
  ).length;
}
```
