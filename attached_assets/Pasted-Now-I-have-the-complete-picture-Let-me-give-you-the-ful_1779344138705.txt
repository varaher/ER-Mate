// Step 1 — Transcribe with Sarvam
const transcribeWithSarvam = async (audioBlob) => {
  const formData = new FormData();
  formData.append('file', audioBlob, 'recording.wav');
  formData.append('model', 'saaras-v2');
  formData.append('language_code', 'unknown'); // auto-detect
  formData.append('with_timestamps', false);
  formData.append('punctuation_mode', 'intelligent');

  const response = await fetch(
    'https://api.sarvam.ai/speech-to-text',
    {
      method: 'POST',
      headers: { 'api-subscription-key': SARVAM_KEY },
      body: formData
    }
  );

  const data = await response.json();
  return {
    transcript: data.transcript,
    detectedLanguage: data.language_code // e.g. "ta-IN", "hi-IN"
  };
};

// Step 2 — Translate to English if needed
const translateToEnglish = async (text, sourceLang) => {
  if (sourceLang === 'en-IN') return text; // already English
  
  const response = await fetch(
    'https://api.sarvam.ai/translate',
    {
      method: 'POST',
      headers: {
        'api-subscription-key': SARVAM_KEY,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        input: text,
        source_language_code: sourceLang,
        target_language_code: 'en-IN',
        speaker_gender: 'Male',
        mode: 'formal',
        enable_preprocessing: true
      })
    }
  );

  const data = await response.json();
  return data.translated_text;
};

// Step 3 — Extract into master schema
const extractClinicalData = async (englishTranscript) => {
  const response = await fetch(
    'https://api.anthropic.com/v1/messages',
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': ANTHROPIC_KEY,
        'anthropic-version': '2023-06-01'
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-20250514',
        max_tokens: 4000,
        messages: [{
          role: 'user',
          content: MASTER_EXTRACTION_PROMPT(englishTranscript)
        }]
      })
    }
  );

  const data = await response.json();
  const raw = data.content[0].text;
  
  // Strip any markdown if model adds it
  const clean = raw
    .replace(/```json/g, '')
    .replace(/```/g, '')
    .trim();
    
  return JSON.parse(clean);
};

// Master orchestrator — never call these separately
const processVoiceCase = async (audioBlob) => {
  try {
    // 1. Transcribe
    const { transcript, detectedLanguage } =

      success: true
    };

const saveCase = async (mergedData) => {
  const token = await getAuthToken();
  
  if (!token) {
    throw new Error('Not authenticated — please log in again');
  }

  // Build complete payload — one object, one call
  const payload = {
    // Patient
    patientName:    mergedData.patient.name,
    patientAge:     mergedData.patient.age,
    patientGender:  mergedData.patient.gender,
    arrivalTime:    mergedData.patient.arrivalTime,
    modeOfArrival:  mergedData.patient.modeOfArrival,
    mlcCase:        mergedData.patient.mlcCase,

    // Doctors
    emResident:     mergedData.doctors.emResident,
    emConsultant:   mergedData.doctors.emConsultant,
    consultations:  mergedData.doctors.consultations,

    // Full clinical data — all sections
    clinicalData: {
      chiefComplaint:  mergedData.chiefComplaint,
      hpi:             mergedData.hpi,
      pastHistory:     mergedData.pastHistory,
      primarySurvey:   mergedData.primarySurvey,
      examination:     mergedData.examination,
      investigations:  mergedData.investigations,
      treatment:       mergedData.treatment,
      diagnosis:       mergedData.diagnosis,
      disposition:     mergedData.disposition,
      psychological:   mergedData.psychological
    },

    // Metadata
    inputMethod:      'voice',
    originalLanguage: mergedData.detectedLanguage,
    originalTranscript: mergedData.transcript
  };

  const response = await fetch('/api/cases', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${token}`
    },
    body: JSON.stringify(payload)
  });

  if (!response.ok) {
    const err = await response.json();
    throw new Error(err.message || 'Save failed');
  }

  return response.json();
};
