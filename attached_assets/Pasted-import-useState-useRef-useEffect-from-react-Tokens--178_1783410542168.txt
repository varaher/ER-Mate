// ── Case note — full hospital EMR format ──────────────────────────────────────
function CaseNoteBlock({ data, onCopy, onEdit, onSave, copied }) {
  const v = data.vitals || {};
  const h = data.history || {};
  const p = data.primary || {};
  const e = data.examination || {};
  const t = data.treatment || {};
  const d = data.disposition || {};
  const psych = h.psychological || {};

  const val = (x, fb = "—") => x && x !== "" ? x : fb;
  const isAbn = (val, field) => {
    const n = parseFloat(val);
    if (isNaN(n)) return false;
    const r = { hr:[60,100], spo2:[95,100], rr:[12,20] };
    const range = r[field];
    if (!range) return false;
    return n < range[0] || n > range[1];
  };

  const row = (label, value, abn = false) => (
    <div style={{ marginBottom: 3 }}>
      <span style={{ color: C.muted, fontSize: 12.5 }}>· {label} → </span>
      <span style={{
        fontSize: 13, color: abn ? C.red : C.ink,
        fontWeight: abn ? 600 : 400,
      }}>{value}{abn ? " ⚠" : ""}</span>
    </div>
  );

  const section = (title, children) => (
    <div style={{ marginBottom: 14 }}>
      <div style={{
        fontSize: 10.5, fontWeight: 800, letterSpacing: 1.2,
        textTransform: "uppercase", color: C.faint,
        marginBottom: 6, paddingBottom: 4,
        borderBottom: `1px solid ${C.border}`,
      }}>{title}</div>
      {children}
    </div>
  );

  const line = (label, value) => (
    <div style={{ marginBottom: 4, display:"flex", gap:8 }}>
      <span style={{ fontSize:12.5, color:C.muted, width:130, flexShrink:0 }}>{label}</span>
      <span style={{ fontSize:13, color:C.ink, lineHeight:1.5 }}>{val(value)}</span>
    </div>
  );

  // General exam toggles → sentence
  const genExam = () => {
    const findings = [];
    if (e.general?.pallor)           findings.push("pallor");
    if (e.general?.icterus)          findings.push("icterus");
    if (e.general?.cyanosis)         findings.push("cyanosis");
    if (e.general?.clubbing)         findings.push("clubbing");
    if (e.general?.lymphadenopathy)  findings.push("lymphadenopathy");
    if (e.general?.edema)            findings.push("edema");
    if (findings.length === 0) {
      return "No pallor, icterus, cyanosis, clubbing, lymphadenopathy, or edema.";
    }
    return `${findings.join(", ")} present.`;
  };

  // Psychological assessment → sentence
  const psychAssess = () => {
    const findings = [];
    if (psych.suicidalIdeation)    findings.push("suicidal ideation");
    if (psych.selfHarmHistory)     findings.push("self-harm history");
    if (psych.substanceAbuse)      findings.push("substance abuse");
    if (psych.psychiatricHistory)  findings.push("psychiatric history");
    if (findings.length === 0) {
      return "No features of depression, anxiety, psychosis, agitation, suicidal ideation, or substance use.";
    }
    return `Features noted: ${findings.join(", ")}. ${val(psych.notes)}`;
  };

  // Differentials with reasoning
  const diffs = d.differentialDiagnoses;

  return (
    <div style={{
      background: C.white,
      borderRadius: 14,
      border: `1px solid ${C.border}`,
      overflow: "hidden",
      boxShadow: `0 2px 12px rgba(13,107,58,0.08)`,
    }}>
      {/* Document header — dark emerald */}
      <div style={{
        background: C.emDeep,
        padding: "12px 16px",
        display:"flex", alignItems:"center", gap:10,
      }}>
        <div style={{ flex:1 }}>
          <div style={{ fontSize:11, fontWeight:800, color:"rgba(255,255,255,0.5)", letterSpacing:1, textTransform:"uppercase" }}>
            Emergency Case Record
          </div>
          <div style={{ fontSize:13, fontWeight:700, color:"white", marginTop:2 }}>
            Initial Assessment & ED Case Note
          </div>
        </div>
        <div style={{
          background: "rgba(29,184,112,0.25)",
          borderRadius:6, padding:"3px 10px",
          fontSize:9, fontWeight:800, color:C.em, letterSpacing:0.8,
        }}>CASE NOTE</div>
      </div>

      {/* Document body */}
      <div style={{ padding:"16px 18px", fontFamily:"Georgia, serif" }}>

        {/* Patient header */}
        <div style={{ marginBottom:14, paddingBottom:12, borderBottom:`1px solid ${C.border}` }}>
          <div style={{ fontSize:15, fontWeight:700, color:C.ink, fontFamily:"inherit", marginBottom:4 }}>
            {val(data.patientName, "Patient")} · {val(data.age, "—")}{data.gender ? data.gender[0] : ""}
          </div>
          <div style={{ display:"flex", gap:8, flexWrap:"wrap" }}>
            <span style={{
              fontSize:10, fontWeight:800,
              color: data.priority === "P1" ? C.red : data.priority === "P2" ? C.amber : C.emDeep,
              background: data.priority === "P1" ? C.redBg : data.priority === "P2" ? C.amberBg : C.emLight,
              borderRadius:5, padding:"3px 9px",
            }}>{val(data.priority, "P4")}</span>
            <span style={{ fontSize:11, color:C.muted }}>
              CASE SEEN BY Dr. {val(data.doctorName, "—")}
            </span>
            <span style={{ fontSize:11, color:C.muted }}>
              {val(data.date)} · {val(data.time)}
            </span>
          </div>
        </div>

        {section("Presenting Complaint",
          <div style={{ fontSize:13.5, color:C.ink, lineHeight:1.6, fontStyle: val(data.chiefComplaint) === "—" ? "italic" : "normal", color: val(data.chiefComplaint) === "—" ? C.faint : C.ink }}>
            {val(data.chiefComplaint, "Not documented")}
          </div>
        )}

        {section("Primary Assessment (ABCDE)",
          <>
            {row("Airway", val(p.airway?.notes, "Patent, self-maintained"))}
            {row("Breathing",
              `RR: ${val(v.rr, "—")}/min · SpO₂: ${val(v.spo2, "—")}% ${p.breathing?.notes ? "· " + p.breathing.notes : "on room air"}`,
              isAbn(v.spo2, "spo2")
            )}
            {row("Circulation",
              `CRT ${val(e.crt, "<2s")} · HR: ${val(v.hr, "—")}/min · BP: ${val(v.bp, "—")} mmHg${p.circulation?.notes ? " · " + p.circulation.notes : ""}`,
              isAbn(v.hr, "hr")
            )}
            {row("Disability",
              `GCS: ${val(v.gcs, "—")} · ${val(p.disability?.notes, "Alert")}`
            )}
            {row("Exposure", `Temp: ${val(v.temp, "—")}`)}
            {(p.adjuncts?.ecg?.performed || p.adjuncts?.abgVbg?.performed || p.adjuncts?.efast?.performed) && (
              <div style={{ marginTop:6, fontSize:12, color:C.muted }}>
                Adjuncts: {[
                  p.adjuncts?.ecg?.performed && `ECG${p.adjuncts.ecg.findings ? " (" + p.adjuncts.ecg.findings + ")" : ""}`,
                  p.adjuncts?.abgVbg?.performed && `VBG/ABG${p.adjuncts.abgVbg.lactate ? " · Lactate " + p.adjuncts.abgVbg.lactate : ""}`,
                  p.adjuncts?.efast?.performed && "EFAST",
                ].filter(Boolean).join(" · ")}
              </div>
            )}
          </>
        )}

        {section("History of Present Illness",
          <div style={{ fontSize:13, color:C.ink, lineHeight:1.7, fontStyle: val(h.events) === "—" ? "italic" : "normal" }}>
            {val(h.events, "History to be documented.")}
          </div>
        )}

        {section("Secondary Survey",
          <>
            {line("Signs & Symptoms", h.signsSymptoms)}
            {line("Past Medical Hx", h.pastMedicalHistory)}
            {line("Surgical Hx", h.pastSurgicalHistory)}
            {line("Family / Gynae Hx", h.familyHistory)}
            {data.gender === "Female" && line("LMP", h.lmp)}
            {line("Allergies", h.allergies || "No known allergies")}
          </>
        )}

        {section("General Examination",
          <div style={{ fontSize:13, color:C.ink, lineHeight:1.6 }}>{genExam()}</div>
        )}

        {section("Systemic Examination",
          <>
            {line("CVS", e.cardiovascular)}
            {line("Chest", e.respiratory)}
            {line("Abdomen", e.abdomen)}
            {line("CNS", e.cns)}
            {line("Extremities", e.extremities)}
          </>
        )}

        {section("Psychological Assessment",
          <div style={{ fontSize:13, color:C.ink, lineHeight:1.6 }}>{psychAssess()}</div>
        )}

        {section("Investigations",
          <>
            {line("Labs ordered", t.investigations?.labsOrdered)}
            {line("Imaging", t.investigations?.imaging)}
            {t.investigations?.resultsSummary && line("Results", t.investigations.resultsSummary)}
          </>
        )}

        {section("Treatment Plan",
          <>
            {Array.isArray(t.medications) && t.medications.length > 0 ? (
              t.medications.map((m, i) => (
                <div key={i} style={{ fontSize:13, color:C.ink, marginBottom:3 }}>
                  · {m.drugName} {m.dose} {m.route} {m.frequency}
                </div>
              ))
            ) : (
              <div style={{ fontSize:13, color:C.faint, fontStyle:"italic" }}>
                {val(t.medications)}
              </div>
            )}
            {t.procedures && <div style={{ fontSize:13, color:C.ink, marginTop:4 }}>Procedures: {t.procedures}</div>}
          </>
        )}

        {section("Disposition",
          <div style={{ fontSize:13, color:C.ink }}>{val(d.dispositionType, "Pending")}</div>
        )}

        {/* DIAGNOSIS SECTION — from case sheet */}
        {section("Provisional Diagnosis",
          <>
            <div style={{ marginBottom:10 }}>
              <div style={{ fontSize:10, fontWeight:700, color:C.muted, letterSpacing:0.8, textTransform:"uppercase", marginBottom:4 }}>
                Primary Diagnosis
              </div>
              <div style={{
                fontSize:14, fontWeight:700,
                color: t.provisionalDiagnosis?.primaryDiagnosis ? C.ink : C.faint,
                fontStyle: t.provisionalDiagnosis?.primaryDiagnosis ? "normal" : "italic",
              }}>
                {val(t.provisionalDiagnosis?.primaryDiagnosis, "Pending — to be documented")}
              </div>
            </div>
            {(diffs || t.provisionalDiagnosis?.differentialDiagnoses) && (
              <div>
                <div style={{ fontSize:10, fontWeight:700, color:C.muted, letterSpacing:0.8, textTransform:"uppercase", marginBottom:6 }}>
                  Differential Diagnoses
                </div>
                {(Array.isArray(diffs) ? diffs : Array.isArray(t.provisionalDiagnosis?.differentialDiagnoses) ? t.provisionalDiagnosis.differentialDiagnoses : typeof t.provisionalDiagnosis?.differentialDiagnoses === "string" ? t.provisionalDiagnosis.differentialDiagnoses.split(",") : [])
                  .filter(Boolean)
                  .map((dd, i) => (
                    <div key={i} style={{
                      display:"flex", gap:8, alignItems:"flex-start",
                      marginBottom:5,
                    }}>
                      <span style={{ color:C.em, fontWeight:700, fontSize:13 }}>•</span>
                      <span style={{ fontSize:13, color:C.ink, lineHeight:1.5 }}>{dd.trim()}</span>
                    </div>
                  ))
                }
              </div>
            )}
          </>
        )}

        {/* Signatures */}
        <div style={{ marginTop:14, paddingTop:12, borderTop:`1px solid ${C.border}` }}>
          <div style={{ display:"flex", justifyContent:"space-between", flexWrap:"wrap", gap:12 }}>
            <div>
              <div style={{ fontSize:10, color:C.faint, textTransform:"uppercase", letterSpacing:0.8 }}>EM Resident</div>
              <div style={{ fontSize:13, fontWeight:600, color:C.ink, marginTop:2 }}>Dr. {val(data.residentName, "_______________")}</div>
            </div>
            <div>
              <div style={{ fontSize:10, color:C.faint, textTransform:"uppercase", letterSpacing:0.8 }}>EM Consultant</div>
              <div style={{ fontSize:13, fontWeight:600, color:C.ink, marginTop:2 }}>Dr. {val(data.consultantName, "_______________")}</div>
            </div>
          </div>
        </div>
      </div>

      {/* Action bar */}
      <div style={{
        borderTop: `1px solid ${C.border}`,
        background: C.emTint,
        padding:"10px 16px",
        display:"flex", gap:8, flexWrap:"wrap",
      }}>
        <button onClick={onCopy} style={{
          display:"flex", alignItems:"center", gap:5,
          background: copied ? C.emLight : C.emDeep,
          border: "none", borderRadius:9,
          padding:"8px 14px",
          fontSize:12, fontWeight:700,
          color: copied ? C.emDeep : "white",
          cursor:"pointer", fontFamily:"inherit",
          transition:"all 0.2s",
        }}>
          <CopyIcon/>
          {copied ? "Copied ✓" : "Copy all"}
        </button>
        <button onClick={onEdit} style={{
          display:"flex", alignItems:"center", gap:5,
          background: C.white, border:`1px solid ${C.border}`,
          borderRadius:9, padding:"8px 14px",
          fontSize:12, fontWeight:700, color:C.muted,
          cursor:"pointer", fontFamily:"inherit",
        }}>
          <EditIcon/>Edit fields
        </button>
        <button onClick={onSave} style={{
          display:"flex", alignItems:"center", gap:5,
          background: C.white, border:`1px solid ${C.emBorder}`,
          borderRadius:9, padding:"8px 14px",
          fontSize:12, fontWeight:700, color:C.emDark,
          cursor:"pointer", fontFamily:"inherit",
        }}>
          💾 Save
        </button>
      </div>

      {/* Feedback */}
      <div style={{
        borderTop:`1px solid ${C.border}`,
        padding:"9px 16px",
        display:"flex", alignItems:"center", gap:10,
      }}>
        <span style={{ fontSize:12, color:C.muted, flex:1 }}>Was this accurate?</span>
        <button style={{
          background:C.white, border:`1px solid ${C.border}`,
          borderRadius:8, width:34, height:34,
          display:"flex", alignItems:"center", justifyContent:"center",
          cursor:"pointer", color:C.muted,
        }}><ThumbUp/></button>
        <button style={{
          background:C.white, border:`1px solid ${C.border}`,
          borderRadius:8, width:34, height:34,
          display:"flex", alignItems:"center", justifyContent:"center",
          cursor:"pointer", color:C.muted,
        }}><ThumbDown/></button>
      </div>
    </div>
  );
}

// ── Discharge summary block ───────────────────────────────────────────────────
function DischargeSummaryBlock({ data, onCopy, copied }) {
  const v = data.vitals || {};
  const h = data.history || {};
  const e = data.examination || {};
  const p = data.primary || {};
  const t = data.treatment || {};
  const d = data.disposition || {};

  const val = (x, fb = "—") => x && x !== "" ? x : fb;

  const line = (label, value) => value && value !== "—" ? (
    <div style={{ marginBottom:4 }}>
      <span style={{ fontSize:12, color:C.muted }}>· {label}: </span>
      <span style={{ fontSize:13, color:C.ink }}>{value}</span>
    </div>
  ) : null;

  return (
    <div style={{
      background:C.white, borderRadius:14,
      border:`1px solid ${C.border}`,
      overflow:"hidden",
      boxShadow:`0 2px 12px rgba(13,107,58,0.08)`,
    }}>
      <div style={{
        background:"linear-gradient(135deg,#1E3A2E,#0D6B3A)",
        padding:"12px 16px",
        display:"flex", alignItems:"center", gap:10,
      }}>
        <div style={{ flex:1 }}>
          <div style={{ fontSize:10.5, fontWeight:800, color:"rgba(255,255,255,0.45)", letterSpacing:1, textTransform:"uppercase" }}>
            Emergency Department
          </div>
          <div style={{ fontSize:13, fontWeight:700, color:"white", marginTop:2 }}>
            Discharge Summary
          </div>
        </div>
        <div style={{
          background:"rgba(29,184,112,0.25)",
          borderRadius:6, padding:"3px 10px",
          fontSize:9, fontWeight:800, color:C.em, letterSpacing:0.8,
        }}>DISCHARGE</div>
      </div>

      <div style={{ padding:"16px 18px" }}>

        <div style={{ marginBottom:14, paddingBottom:12, borderBottom:`1px solid ${C.border}` }}>
          <div style={{ fontSize:15, fontWeight:700, color:C.ink, marginBottom:4 }}>
            {val(data.patientName, "Patient")} · {val(data.age)}yr · {val(data.gender)}
          </div>
          <div style={{ fontSize:11, color:C.muted }}>
            Date: {val(data.date)} · {val(data.department, "Emergency Department")} · Dr. {val(data.doctorName)}
          </div>
        </div>

        {/* Section 1 */}
        <div style={{ marginBottom:14 }}>
