// ── Single source of truth — feature access ────────────────────────────────
function getAccess(sub) {
  const unlimited = sub.teamActive || sub.plan === "pro";
  const isFree = !unlimited;
  return {
    unlimited,
    isFree,
    cases:           unlimited ? "unlimited" : `${sub.casesUsed || 0} of 10`,
    smartDictation:  true,   // always free on any plan
    discharge:       true,   // always free on any plan
    aiFeatures:      unlimited ? "unlimited" : (sub.aiCredits > 0 ? "credits" : "blocked"),
    aiCredits:       sub.aiCredits || 0,
    rounds:          unlimited ? "unlimited" : (sub.aiCredits > 0 ? "credits" : "blocked"),
    clinicalMemory:  sub.plan === "pro", // private only with Pro
    handover:        sub.teamActive,
    shiftMgmt:       sub.teamActive,
    hod:             sub.teamActive && sub.teamRole === "hod",
    showCreditBar:   isFree,
    showUpgrade:     isFree,
  };
}

// ── Credit bar — Free users only ───────────────────────────────────────────
function CreditBar({ credits, maxCredits=5, onUpgrade }) {
  const used = maxCredits - credits;
  const pct  = (used / maxCredits) * 100;
  const low  = credits <= 2;
  const gone = credits === 0;

  return (
    <div style={{
      background: gone ? "#FEF2F2" : low ? C.orangeLight : C.greenLight,
      border: `1px solid ${gone ? "#FECACA" : low ? "rgba(245,158,11,0.2)" : C.greenBorder}`,
      borderRadius:14, padding:"13px 14px",
    }}>
      <div style={{ display:"flex", alignItems:"center", justifyContent:"space-between", marginBottom:8 }}>
        <div style={{ fontSize:12, fontWeight:700,
          color:gone ? C.red : low ? "#92400E" : C.greenDark }}>
          {gone ? "AI credits used up" : `${credits} AI credit${credits===1?"":"s"} remaining`}
        </div>
        <div style={{ fontSize:11, color:C.faint }}>{used}/{maxCredits} used</div>
      </div>
      <div style={{ height:6, background:"rgba(0,0,0,0.06)", borderRadius:99, overflow:"hidden", marginBottom:10 }}>
        <div style={{
          height:"100%", borderRadius:99,
          width:`${pct}%`,
          background: gone ? C.red : low ? C.orange : C.green,
          transition:"width 0.5s ease",
        }}/>
      </div>
      <div style={{ fontSize:11, color:C.muted, lineHeight:1.5, marginBottom:gone||low?10:0 }}>
        {gone
          ? "Upgrade to Pro for unlimited AI — Decision Support, Rounds, OCR, and more."
          : low
          ? `${credits} credit${credits===1?"":"s"} left. Each AI action uses 1 credit.`
          : "Each AI action (Decision Support, Rounds, OCR) uses 1 credit."}
      </div>
      {(gone || low) && (
        <button onClick={onUpgrade} style={{
          width:"100%", background:`linear-gradient(135deg,${C.green},${C.greenDark})`,
          border:"none", borderRadius:10, padding:"10px",
          fontSize:13, fontWeight:700, color:"white", cursor:"pointer",
          fontFamily:"inherit", marginTop:2,
        }}>
          Upgrade to Pro — Unlimited →
        </button>
      )}
    </div>
  );
}
