// ─────────────────────────────────────────────────────────────────────────────
// HOW TO USE IN CaseChatScreen.tsx
// ─────────────────────────────────────────────────────────────────────────────
//
// 1. Import at top of CaseChatScreen.tsx:
//    import { chatStyles, Colors, Type, Space } from '../styles/caseChatTheme';
//
// 2. Replace every StyleSheet reference:
//    OLD: styles.container  → NEW: chatStyles.screen
//    OLD: styles.header     → NEW: chatStyles.header
//    OLD: styles.chatBg     → NEW: chatStyles.chatScroll
//    etc.
//
// 3. Key colour replacements:
//    '#0F1419' (dark bg)    → Colors.bgChat
//    '#1A2332' (bubble)     → Colors.bgDoctorBubble
//    'rgba(255,255,255,0.x) → use Colors.textMuted / Colors.textFaint
//    '#1DB870'              → Colors.emerald (same value, keeps consistency)
//    '#0B0F14' (doc header) → Colors.docHeaderBg
//    'white' input bg       → Colors.bgInput
//
// 4. Header background:
//    OLD: '#0B0F14' or '#1A2332'
//    NEW: Colors.emeraldDark  ('#15924F')
