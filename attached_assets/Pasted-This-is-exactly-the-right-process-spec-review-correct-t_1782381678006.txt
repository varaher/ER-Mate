```typescript
// Write this test endpoint in routes.ts
// GET /api/test/merge/:caseId

app.get("/api/test/merge/:caseId", async (req, res) => {
  // 1. Fetch from external backend
  const externalCase = await fetch(
    `https://er-emr-backend.onrender.com/api/cases/${req.params.caseId}`,
    { headers: { Authorization: req.headers.authorization } }
  ).then(r => r.json());

  // 2. Fetch local overlay
  const overlay = await db
    .select()
    .from(caseOverlays)
    .where(eq(caseOverlays.caseId, req.params.caseId))
    .limit(1);

  // 3. Merge and return
  res.json({
    ...externalCase,
    overlay: overlay[0] || null,
    merged: true,
  });
});

// Expected: returns case data with overlay: null
// (since no overlay exists yet)
// If this returns 200 — merge pattern works. Proceed.
```
