      <Text style={styles.body}>

        SpO₂: {val(v.spo2)} · RR: {val(v.rr)} · 
        Temp: {val(v.temp)} · GCS: {val(v.gcs)} · 
        GRBS: {val(v.grbs)}
      </Text>

      <Text style={styles.body}>

        Allergies: {val(h.allergies, 'NKDA')}{'\n'}
        Medications: {val(h.medications, 'None')}{'\n'}
        Past Hx: {val(h.pastHistory, 'Nil significant')}{'\n'}
        Last meal: {val(h.lastMeal)}{'\n'}
        Events: {val(h.events)}
      </Text>
      
      <Text style={styles.section}>
        PRIMARY SURVEY (ABCDE)
      </Text>
      <Text style={styles.body}>
        A: {val(p.airway, 'Patent')}{'\n'}
        B: {val(p.breathing)}{'\n'}
        C: {val(p.circulation)}{'\n'}
        D: {val(p.disability)}{'\n'}

        {val(d.diagnosis, 'Pending')}

        {val(d.plan)}{'\n'}

// Fix the "Not captured" banner
// Only flag fields that are truly empty
function getMissing(data) {
  const missing = [];
  const v = data.vitals || {};
  const h = data.history || {};
  const d = data.disposition || {};
  
  // Only flag if truly absent
  if (!data.patientName) missing.push('Name');
  if (!h.symptoms && !data.chiefComplaint) 
    missing.push('Chief complaint');
  if (!d.diagnosis) missing.push('Impression');
  // DO NOT flag HR/BP/SpO₂ if they have values
  
  return missing;
}
