## What Needs to Be Built — Backend

```typescript
// The rota_assignments table already has a "date" field
// Querying is just a date-range filter:

GET /api/department/:id/rota?startDate=2026-07-13&endDate=2026-07-19

// Backend logic:
async function getRotaForWeek(deptId, startDate, endDate) {
  return await db.select()
    .from(rotaAssignments)
    .where(
      and(
        eq(rotaAssignments.departmentId, deptId),
        gte(rotaAssignments.date, startDate),
        lte(rotaAssignments.date, endDate)
      )
    );
}

// No special "reset" logic needed —
// each week is just a different date range
// queried from the same table
```

## What Needs to Be Built — Frontend

```typescript
// In ManageRotaScreen.tsx

const [weekStart, setWeekStart] = useState(getMonday(new Date()));

function getMonday(date) {
  const d = new Date(date);
  const day = d.getDay();
  const diff = d.getDate() - day + (day === 0 ? -6 : 1);
  return new Date(d.setDate(diff));
}

function getWeekDates(monday) {
  return Array.from({ length: 7 }, (_, i) => {
    const d = new Date(monday);
    d.setDate(d.getDate() + i);
    return d;
  });
}

// When ‹ or › tapped:
const goToPrevWeek = () => {
  const prev = new Date(weekStart);
  prev.setDate(prev.getDate() - 7);
  setWeekStart(prev);
  fetchRota(prev); // re-fetch from backend for new range
};

const goToNextWeek = () => {
  const next = new Date(weekStart);
  next.setDate(next.getDate() + 7);
  setWeekStart(next);
  fetchRota(next);
};

// On mount and whenever weekStart changes:
useEffect(() => {
  fetchRota(weekStart);
}, [weekStart]);
```
