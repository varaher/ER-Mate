**Tell your developer to add this check first — before the fetch call:**

```javascript
const handleExtract = async () => {
  
  // Step 1 — Check what we're sending
  const textToExtract = englishTranscript 
    || editedTranscript;
  
  console.log('Text to extract:', 
    textToExtract?.substring(0, 100));
  console.log('Text length:', 
    textToExtract?.length);
  
  // If empty — stop immediately
  if (!textToExtract || 
      textToExtract.trim().length < 10) {
    Alert.alert(
      'Nothing to extract',
      'Transcript is empty. Please re-record.'
    );
    return;
  }
  
  // Step 2 — Check auth token
  const token = await getAuthToken();
  console.log('Token exists:', !!token);
  
  if (!token) {
    Alert.alert(
      'Session expired',
      'Please log in again.'
    );
    return;
  }
  
  // Step 3 — Make the call with timeout
  setIsExtracting(true);
  
  const controller = new AbortController();
  const timeout = setTimeout(() => {
    controller.abort();
  }, 40000);
  
  try {
    console.log('Sending to API...');
    
    const response = await fetch(
      '/api/voice/smart-dictation',
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({ 
          transcript: textToExtract 
        }),
        signal: controller.signal
      }
    );
    
    clearTimeout(timeout);
    console.log('Response received:', 
      response.status);
    
    const data = await response.json();
    console.log('Data keys:', 
      Object.keys(data));
    
    if (!response.ok) {
      throw new Error(
        data.error || `Server error ${response.status}`
      );
    }
    
    // Set data and move to review
    setExtractedData(data);
    setStep('review');
    
  } catch (error) {
    clearTimeout(timeout);
    
    if (error.name === 'AbortError') {
      Alert.alert(
        'Taking too long',
        'The extraction timed out. ' +
        'Check your connection and try again.'
      );
    } else {
      Alert.alert(
        'Extraction failed',
        error.message
      );
    }
    
  } finally {
    setIsExtracting(false);
  }
};
```
