        <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
          {THINKING_MODES.map((mode, i) => (
            <button
              key={mode.id}
              onClick={() => startDebrief(mode)}
              style={{
                background: mode.id === "full_debrief" ? "linear-gradient(135deg, #0D2E1A, #0a2216)" : "white",
                border: mode.id === "full_debrief" ? "1.5px solid rgba(30,184,112,0.3)" : "1.5px solid #F0F0F0",
                borderRadius: 16, padding: "14px 16px",
                display: "flex", alignItems: "center", gap: 14,
                cursor: "pointer", textAlign: "left",
                animation: `fadeUp 0.4s ${i*0.05}s ease both`,
                transition: "transform 0.15s",
              }}
              onMouseEnter={e => e.currentTarget.style.transform = "translateX(4px)"}
              onMouseLeave={e => e.currentTarget.style.transform = "none"}
            >
              <div style={{
                width: 44, height: 44, borderRadius: 13,
                background: mode.colorBg,
                border: `1.5px solid ${mode.color}33`,
                display: "flex", alignItems: "center", justifyContent: "center",
                fontSize: 22, flexShrink: 0,
              }}>
                {mode.emoji}
              </div>
              <div style={{ flex: 1 }}>
                <div style={{
                  fontSize: 15, fontWeight: 700,
                  color: mode.id === "full_debrief" ? "white" : "#0D1117",
                  marginBottom: 3,
                }}>{mode.label}</div>
                <div style={{
                  fontSize: 12,
                  color: mode.id === "full_debrief" ? "rgba(255,255,255,0.45)" : "#9CA3AF",
                }}>{mode.tagline}</div>
              </div>
              <div style={{ color: mode.id === "full_debrief" ? "rgba(255,255,255,0.3)" : "#C4C9D4" }}>
                <IconChevron right />
              </div>
            </button>
          ))}
        </div>
      </div>
    </div>
  );

  // ── DEBRIEF VIEW ───────────────────────────────────────────────────────
  const DebriefView = () => (
    <div style={{ paddingBottom: 80 }}>
      {/* Header */}
      <div style={{
        background: selectedMode?.colorBg || "rgba(30,184,112,0.08)",
        borderBottom: `2px solid ${selectedMode?.color || "#1DB870"}22`,
        padding: "20px 20px 18px",
      }}>
        <button onClick={() => setView("case")} style={{ background: "none", border: "none", cursor: "pointer", color: "#6B7280", display: "flex", alignItems: "center", gap: 6, marginBottom: 14, padding: 0, fontSize: 13 }}>
          <IconBack /> Back
        </button>
        <div style={{ display: "flex", alignItems: "center", gap: 12 }}>
          <div style={{
            width: 48, height: 48, borderRadius: 14,
            background: selectedMode?.colorBg,
            border: `2px solid ${selectedMode?.color}33`,
            display: "flex", alignItems: "center", justifyContent: "center",
            fontSize: 24,
          }}>
            {selectedMode?.emoji}
          </div>
          <div>
            <div style={{ fontSize: 16, fontWeight: 800, color: "#0D1117" }}>{selectedMode?.label}</div>
            <div style={{ fontSize: 12, color: "#9CA3AF", marginTop: 2 }}>
              {selectedCase?.age}{selectedCase?.gender} · {selectedCase?.complaint}
            </div>
          </div>
        </div>
      </div>

      {/* Debrief content */}
      <div style={{ padding: "20px 20px 0" }}>
        {loading && !debriefText && <LoadingDots />}
        {debriefText && (
          <div style={{ animation: "fadeUp 0.3s ease" }}>
            {renderMarkdown(debriefText)}
            {loading && <LoadingDots />}
          </div>
        )}
        {!loading && debriefText && (
          <div style={{ marginTop: 24, display: "flex", flexDirection: "column", gap: 10 }}>
            <div style={{
              background: "rgba(30,184,112,0.06)", border: "1px solid rgba(30,184,112,0.15)",
              borderRadius: 14, padding: "12px 16px",
              display: "flex", alignItems: "center", gap: 8,
            }}>
              <IconSparkle />
              <span style={{ fontSize: 13, color: "#15924F", fontWeight: 500 }}>
                Saved to your clinical memory
              </span>
            </div>
            <button
              onClick={() => setView("case")}
              style={{
                background: "#0D1117", color: "white", border: "none",
                borderRadius: 14, padding: "14px", fontSize: 15, fontWeight: 700,
                cursor: "pointer", fontFamily: "inherit",
              }}
            >
              Try another lens →
            </button>
          </div>
        )}
      </div>
    </div>
  );

  // ── MEMORY VIEW ────────────────────────────────────────────────────────
  const MemoryView = () => (
    <div style={{ paddingBottom: 80 }}>
      <div style={{ background: "#0D1117", padding: "20px 20px 28px" }}>
        <button onClick={() => setView("home")} style={{ background: "none", border: "none", cursor: "pointer", color: "rgba(255,255,255,0.5)", display: "flex", alignItems: "center", gap: 6, marginBottom: 16, padding: 0, fontSize: 13 }}>
          <IconBack /> Back
        </button>
        <div style={{ fontSize: 11, color: "rgba(255,255,255,0.3)", fontWeight: 700, letterSpacing: 1.2, textTransform: "uppercase", marginBottom: 8 }}>Clinical Memory</div>
        <div style={{ fontSize: 28, fontWeight: 800, color: "white", letterSpacing: -0.6 }}>{MEMORY_STATS.totalCases} cases</div>
        <div style={{ fontSize: 14, color: "rgba(255,255,255,0.4)", marginTop: 4 }}>documented across your career with ErMate</div>

        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr 1fr", gap: 10, marginTop: 20 }}>
          {[
            { label: "This month", value: MEMORY_STATS.thisMonth },
            { label: "Debriefed", value: MEMORY_STATS.debriefed },
            { label: "Day streak", value: `${MEMORY_STATS.streak}🔥` },
          ].map((s, i) => (
            <div key={i} style={{ background: "rgba(255,255,255,0.05)", borderRadius: 12, padding: "12px 14px" }}>
              <div style={{ fontSize: 20, fontWeight: 800, color: "white" }}>{s.value}</div>
              <div style={{ fontSize: 11, color: "rgba(255,255,255,0.35)", marginTop: 3 }}>{s.label}</div>
            </div>
          ))}
        </div>
      </div>

      <div style={{ padding: "20px 20px 0" }}>
        <div style={{ fontSize: 11, fontWeight: 700, color: "#9CA3AF", letterSpacing: 1.2, textTransform: "uppercase", marginBottom: 14 }}>
          Cases by category
        </div>
        <div style={{ display: "flex", flexDirection: "column", gap: 12 }}>
          {MEMORY_STATS.categories.map((cat, i) => (
            <div key={i}>
              <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 6 }}>
                <span style={{ fontSize: 14, fontWeight: 500, color: "#0D1117" }}>{cat.label}</span>
                <span style={{ fontSize: 14, fontWeight: 700, color: "#0D1117" }}>{cat.count}</span>
              </div>
              <div style={{ height: 8, background: "#F0F0F0", borderRadius: 99, overflow: "hidden" }}>
                <div style={{
                  height: "100%", borderRadius: 99,
                  background: cat.color,
                  width: `${(cat.count / MEMORY_STATS.totalCases) * 100}%`,
                  transition: "width 1s ease",
                }}/>
              </div>
            </div>
          ))}
        </div>

        <div style={{
          background: "#F7F9F7", border: "1.5px solid #E5E7EB",
          borderRadius: 16, padding: "16px 18px", marginTop: 24,
        }}>
          <div style={{ fontSize: 13, fontWeight: 700, color: "#0D1117", marginBottom: 4 }}>Your strongest area</div>
          <div style={{ fontSize: 13, color: "#6B7280" }}>
            <span style={{ color: "#1DB870", fontWeight: 700 }}>Trauma</span> — {MEMORY_STATS.topCategoryPct}% of your cases. You've managed {MEMORY_STATS.categories[0].count} trauma patients with ErMate.
          </div>
        </div>
      </div>
    </div>
  );

  // ── Shell ──────────────────────────────────────────────────────────────
  return (
    <div style={{
      fontFamily: "'DM Sans', 'Helvetica Neue', sans-serif",
      background: "#F7F9F7",
      minHeight: "100vh",
      maxWidth: 430,
      margin: "0 auto",
      position: "relative",
      overflowX: "hidden",
    }}>
      {/* Screen content */}
      <div style={{ minHeight: "calc(100vh - 64px)", overflowY: "auto" }}>
        {view === "home" && <HomeView />}
        {view === "case" && selectedCase && <CaseView />}
        {view === "debrief" && selectedCase && selectedMode && <DebriefView />}
        {view === "memory" && <MemoryView />}
      </div>

      {/* Bottom nav */}
      <div style={{
        position: "fixed", bottom: 0, left: "50%", transform: "translateX(-50%)",
        width: "100%", maxWidth: 430,
        background: "white", borderTop: "1px solid #EFEFEF",
        display: "grid", gridTemplateColumns: "repeat(4, 1fr)",
        padding: "8px 0 20px",
      }}>
        {[
          { id: "home", emoji: "🏠", label: "Home" },
          { id: "cases", emoji: "📋", label: "Cases" },
          { id: "rounds", emoji: "🧠", label: "Rounds" },
          { id: "profile", emoji: "👤", label: "Profile" },
        ].map(tab => (
          <button
            key={tab.id}
            onClick={() => setActiveTab(tab.id)}
            style={{
              background: "none", border: "none", cursor: "pointer",
              display: "flex", flexDirection: "column", alignItems: "center", gap: 3,
              padding: "4px 0",
            }}
          >
            <span style={{ fontSize: 20 }}>{tab.emoji}</span>
            <span style={{
              fontSize: 10, fontWeight: 600,
              color: activeTab === tab.id ? "#1DB870" : "#C4C9D4",
              letterSpacing: 0.3,
            }}>{tab.label}</span>
            {activeTab === tab.id && (
              <div style={{ width: 4, height: 4, borderRadius: "50%", background: "#1DB870" }}/>
            )}
          </button>
        ))}
      </div>
    </div>
  );
}
