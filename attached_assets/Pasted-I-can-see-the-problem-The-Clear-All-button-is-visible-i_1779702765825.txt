The earlier server log showed the case object has this structure:


The delete function is probably calling:

## Quick Fix

```javascript
// In the delete handler
const handleDeleteCase = async (caseItem) => {
  // Log what ID we have
  console.log('Deleting case:', 
    caseItem.id, 
    caseItem._id,
    caseItem.case_id);
  
  // Try all possible ID field names
  const caseId = caseItem.id 
    || caseItem._id 
    || caseItem.case_id;
  
  if (!caseId) {
    Alert.alert('Error', 
      'Cannot find case ID to delete');
    return;
  }
  
  const token = await getAuthToken();
  
  const response = await fetch(
    `/api/proxy/cases/${caseId}`,
    {
      method: 'DELETE',
      headers: {
        'Authorization': `Bearer ${token}`
      }
    }
  );
  
  if (response.ok) {
    // Remove from local list
    setCases(prev => 
      prev.filter(c => 
        (c.id || c._id) !== caseId
      )
    );
  } else {
    Alert.alert('Delete failed', 
      `Status: ${response.status}`);
  }
};
```
