// ── Cell content — shows assigned doctors ───────────────────────────────────
function RotaCell({ shiftId, day, assignedIds, onTap, isToday, isPast }) {
  const people = assignedIds.map(id => TEAM.find(p => p.id === id)).filter(Boolean);
  const isEmpty = people.length === 0;

  return (
    <button
      onClick={() => onTap(shiftId, day)}
      style={{
        width: "100%", minHeight: 58,
        background: isToday ? C.greenLight : C.white,
        border: `1.5px solid ${isToday ? C.greenBorder : C.border}`,
        borderRadius: 10, padding: "6px 5px",
        cursor: "pointer", fontFamily: "inherit",
        opacity: isPast ? 0.55 : 1,
        display: "flex", flexDirection: "column", gap: 3,
        alignItems: "center", justifyContent: "center",
        transition: "all 0.15s",
      }}
    >
      {isEmpty ? (
        <div style={{
          width: 22, height: 22, borderRadius: "50%",
          border: `1.5px dashed ${C.border}`,
          display: "flex", alignItems: "center", justifyContent: "center",
        }}>
          <PlusIcon color={C.faint} />
        </div>
      ) : (
        <div style={{ display: "flex", gap: -4 }}>
          {people.slice(0, 2).map((p, i) => (
            <div key={p.id} style={{
              width: 22, height: 22, borderRadius: "50%",
              background: p.color, color: "white",
              fontSize: 9, fontWeight: 800,
              display: "flex", alignItems: "center", justifyContent: "center",
              marginLeft: i > 0 ? -6 : 0,
              border: "1.5px solid white",
            }}>{p.initial}</div>
          ))}
          {people.length > 2 && (
            <div style={{
              width: 22, height: 22, borderRadius: "50%",
              background: C.surface, color: C.muted,
              fontSize: 8, fontWeight: 800,
              display: "flex", alignItems: "center", justifyContent: "center",
              marginLeft: -6, border: "1.5px solid white",
            }}>+{people.length - 2}</div>
          )}
        </div>
      )}
    </button>
  );
}

// ── Assignment sheet — slides up when tapping a cell ────────────────────────
function AssignSheet({ shift, day, date, assignedIds, onAssign, onRemove, onClose }) {
  const assigned = assignedIds.map(id => TEAM.find(p => p.id === id)).filter(Boolean);
  const available = TEAM.filter(p => !assignedIds.includes(p.id));
  const shiftData = SHIFTS.find(s => s.id === shift);

  return (
    <div style={{
      position: "absolute", inset: 0, background: "rgba(0,0,0,0.5)",
      display: "flex", alignItems: "flex-end", zIndex: 50,
    }} onClick={onClose}>
      <div onClick={e => e.stopPropagation()} style={{
        background: C.white, borderRadius: "20px 20px 0 0",
        padding: "20px 18px 28px", width: "100%", maxHeight: "75%", overflowY: "auto",
      }}>
        <div style={{ width: 36, height: 4, background: C.border, borderRadius: 99, margin: "0 auto 16px" }}/>

        <div style={{ display: "flex", alignItems: "center", gap: 10, marginBottom: 4 }}>
          <div style={{
            width: 8, height: 8, borderRadius: "50%",
            background: shiftData.color,
          }}/>
          <span style={{ fontSize: 16, fontWeight: 800, color: C.ink }}>
            {shiftData.label} · {day}
          </span>
        </div>
        <div style={{ fontSize: 12, color: C.faint, marginBottom: 18 }}>
          {shiftData.time} · {date}
        </div>

        {/* Currently assigned */}
        {assigned.length > 0 && (
          <div style={{ marginBottom: 18 }}>
            <div style={{ fontSize: 10, fontWeight: 700, color: C.faint, letterSpacing: 0.8,
              textTransform: "uppercase", marginBottom: 10 }}>
              Assigned ({assigned.length})
            </div>
            {assigned.map(p => (
              <div key={p.id} style={{
                display: "flex", alignItems: "center", gap: 10,
                background: C.surface, borderRadius: 11, padding: "10px 12px", marginBottom: 8,
              }}>
                <div style={{
                  width: 32, height: 32, borderRadius: "50%",
                  background: p.color, color: "white", fontSize: 13, fontWeight: 800,
                  display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0,
                }}>{p.initial}</div>
                <div style={{ flex: 1 }}>
                  <div style={{ fontSize: 13, fontWeight: 700, color: C.ink }}>{p.name}</div>
                  <div style={{ fontSize: 11, color: C.faint, textTransform: "capitalize" }}>{p.role}</div>
                </div>
                <button onClick={() => onRemove(p.id)} style={{
                  background: C.redLight, border: "none", borderRadius: 8,
                  width: 28, height: 28, cursor: "pointer",
                  display: "flex", alignItems: "center", justifyContent: "center",
                }}>
                  <CloseIcon color={C.red}/>
                </button>
              </div>
            ))}

        )}

        <div>
          <div style={{ fontSize: 10, fontWeight: 700, color: C.faint, letterSpacing: 0.8,
            textTransform: "uppercase", marginBottom: 10 }}>
            Add to this shift
          </div>
          {available.length === 0 ? (
            <div style={{ fontSize: 12, color: C.faint, padding: "12px 0", textAlign: "center" }}>
              All team members are already assigned
            </div>
          ) : available.map(p => (
            <button key={p.id} onClick={() => onAssign(p.id)} style={{
              width: "100%", display: "flex", alignItems: "center", gap: 10,
              background: C.white, border: `1.5px solid ${C.border}`, borderRadius: 11,
              padding: "10px 12px", marginBottom: 8, cursor: "pointer", fontFamily: "inherit",
              textAlign: "left",
            }}>
              <div style={{
                width: 32, height: 32, borderRadius: "50%",
                background: p.color, color: "white", fontSize: 13, fontWeight: 800,
                display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0,
              }}>{p.initial}</div>
              <div style={{ flex: 1 }}>
                <div style={{ fontSize: 13, fontWeight: 700, color: C.ink }}>{p.name}</div>
                <div style={{ fontSize: 11, color: C.faint, textTransform: "capitalize" }}>{p.role}</div>
              </div>
              <div style={{
                background: C.greenLight, borderRadius: 8, padding: "5px 11px",
                fontSize: 11, fontWeight: 700, color: C.greenDark,
              }}>Assign</div>
            </button>
          ))}
        </div>
      </div>
    </div>
  );
}

// SCREEN 1 — HOD: Manage Rota (week grid)
// ══════════════════════════════════════════════════════════════════════════
function ManageRotaScreen({ onBack, onSwitchView }) {
  const [rota, setRota] = useState(initialRota);
  const [sheet, setSheet] = useState(null); // { shiftId, day }
  const [weekOffset, setWeekOffset] = useState(0);

  const handleCellTap = (shiftId, day) => {
    setSheet({ shiftId, day });
  };

  const handleAssign = (personId) => {
    const key = `${sheet.day}-${sheet.shiftId}`;
    setRota(prev => ({
      ...prev,
      [key]: [...(prev[key] || []), personId],
    }));
  };

  const handleRemove = (personId) => {
    const key = `${sheet.day}-${sheet.shiftId}`;
    setRota(prev => ({
      ...prev,
      [key]: (prev[key] || []).filter(id => id !== personId),
    }));
  };

  const openSlots = Object.values(rota).filter(arr => arr.length === 0).length;
  const totalSlots = DAYS.length * SHIFTS.length;

  return (
    <div style={{ background: C.surface, height: "100%", display: "flex", flexDirection: "column", position: "relative" }}>

      {/* Header */}
      <div style={{
        background: C.white, borderBottom: `1px solid ${C.border}`,
        padding: "14px 16px", display: "flex", alignItems: "center", gap: 10,
      }}>
        <button onClick={onBack} style={{ background: "none", border: "none", cursor: "pointer", padding: 0, display: "flex" }}><Back/></button>
        <div style={{ flex: 1 }}>
          <div style={{ fontSize: 16, fontWeight: 800, color: C.ink }}>Manage Rota</div>
          <div style={{ fontSize: 11, color: C.faint }}>City Emergency · {openSlots} of {totalSlots} slots open</div>
        </div>
        <button onClick={onSwitchView} style={{
          background: C.greenLight, border: `1px solid ${C.greenBorder}`,
          borderRadius: 9, padding: "7px 12px", fontSize: 11, fontWeight: 700,
          color: C.greenDark, cursor: "pointer", fontFamily: "inherit",
        }}>My View</button>
      </div>

      {/* Week nav */}
      <div style={{
        background: C.white, borderBottom: `1px solid ${C.border}`,
        padding: "10px 16px", display: "flex", alignItems: "center", justifyContent: "space-between",
      }}>
        <button onClick={() => setWeekOffset(w => w - 1)} style={{
          background: C.surface, border: "none", borderRadius: 8, width: 30, height: 30,
          cursor: "pointer", fontSize: 14, color: C.muted,
        }}>‹</button>
        <span style={{ fontSize: 13, fontWeight: 700, color: C.ink }}>
          {weekOffset === 0 ? "This Week" : weekOffset > 0 ? `${weekOffset} week${weekOffset>1?"s":""} ahead` : `${-weekOffset} week${weekOffset<-1?"s":""} ago`}
          {" · Jul 13–19"}
        </span>
        <button onClick={() => setWeekOffset(w => w + 1)} style={{
          background: C.surface, border: "none", borderRadius: 8, width: 30, height: 30,
          cursor: "pointer", fontSize: 14, color: C.muted,
        }}>›</button>
      </div>

      {/* Grid */}
      <div style={{ flex: 1, overflow: "auto", padding: "14px 10px" }}>
        <div style={{ display: "grid", gridTemplateColumns: "44px repeat(7, 1fr)", gap: 5, minWidth: 560 }}>

          {/* Day headers */}
          <div></div>
          {DAYS.map((d, i) => (
            <div key={d} style={{ textAlign: "center", paddingBottom: 6 }}>
              <div style={{ fontSize: 10, fontWeight: 700, color: i === 1 ? C.green : C.faint }}>{d}</div>
              <div style={{ fontSize: 13, fontWeight: 800, color: i === 1 ? C.green : C.inkSoft }}>{DAY_DATES[i]}</div>
            </div>
          ))}

          {/* Shift rows */}
          {SHIFTS.map(shift => (
            <>
              <div key={shift.id + "-label"} style={{
                display: "flex", flexDirection: "column", alignItems: "center", justifyContent: "center",
                gap: 3,
              }}>
                <div style={{ width: 7, height: 7, borderRadius: "50%", background: shift.color }}/>
                <span style={{ fontSize: 8.5, fontWeight: 700, color: C.muted, writingMode: "horizontal-tb" }}>
                  {shift.label.slice(0,3)}
                </span>
              </div>
              {DAYS.map((day, i) => (
                <RotaCell
                  key={`${day}-${shift.id}`}
                  shiftId={shift.id}
                  day={day}
                  assignedIds={rota[`${day}-${shift.id}`] || []}
                  onTap={handleCellTap}
                  isToday={i === 1}
                  isPast={i === 0}
                />
              ))}
            </>
          ))}
        </div>
      </div>

      {/* Legend */}
      <div style={{
        background: C.white, borderTop: `1px solid ${C.border}`,
        padding: "10px 16px", display: "flex", gap: 16, justifyContent: "center",
      }}>
        {SHIFTS.map(s => (
          <div key={s.id} style={{ display: "flex", alignItems: "center", gap: 5 }}>
            <div style={{ width: 7, height: 7, borderRadius: "50%", background: s.color }}/>
            <span style={{ fontSize: 10, color: C.muted }}>{s.label} {s.time}</span>
          </div>
        ))}
      </div>

      {/* Assignment sheet */}
      {sheet && (
        <AssignSheet
          shift={sheet.shiftId}
          day={sheet.day}
          date={`Jul ${DAY_DATES[DAYS.indexOf(sheet.day)]}`}
          assignedIds={rota[`${sheet.day}-${sheet.shiftId}`] || []}
          onAssign={handleAssign}
          onRemove={handleRemove}
          onClose={() => setSheet(null)}
        />
      )}
    </div>
  );
}

// SCREEN 2 — DOCTOR: My Rota (claim open shifts)
// ══════════════════════════════════════════════════════════════════════════
function MyRotaScreen({ onBack, onSwitchView }) {
  const [rota, setRota] = useState(initialRota);
  const myId = 1; // Dr. Priya, viewing as a resident

  const handleClaim = (key) => {
    setRota(prev => ({ ...prev, [key]: [...(prev[key] || []), myId] }));
  };
  const handleUnclaim = (key) => {
    setRota(prev => ({ ...prev, [key]: (prev[key] || []).filter(id => id !== myId) }));
  };

  const myShifts = [];
  const openShifts = [];
  Object.entries(rota).forEach(([key, ids]) => {
    const [day, shiftId] = key.split("-");
    const shift = SHIFTS.find(s => s.id === shiftId);
    if (ids.includes(myId)) {
      myShifts.push({ key, day, shift });
    } else if (ids.length === 0) {
      openShifts.push({ key, day, shift });
    }
  });

  return (
    <div style={{ background: C.surface, height: "100%", display: "flex", flexDirection: "column" }}>

      <div style={{
        background: C.white, borderBottom: `1px solid ${C.border}`,
        padding: "14px 16px", display: "flex", alignItems: "center", gap: 10,
      }}>
        <button onClick={onBack} style={{ background: "none", border: "none", cursor: "pointer", padding: 0, display: "flex" }}><Back/></button>
        <div style={{ flex: 1 }}>
          <div style={{ fontSize: 16, fontWeight: 800, color: C.ink }}>My Rota</div>
          <div style={{ fontSize: 11, color: C.faint }}>City Emergency · Dr. Priya · Resident</div>
        </div>
        <button onClick={onSwitchView} style={{
          background: C.greenLight, border: `1px solid ${C.greenBorder}`,
          borderRadius: 9, padding: "7px 12px", fontSize: 11, fontWeight: 700,
          color: C.greenDark, cursor: "pointer", fontFamily: "inherit",
        }}>HOD View</button>
      </div>

      <div style={{ flex: 1, overflow: "auto", padding: "16px" }}>

        {/* My upcoming shifts */}
        <div style={{ fontSize: 10, fontWeight: 700, color: C.faint, letterSpacing: 0.8,
          textTransform: "uppercase", marginBottom: 10 }}>
          My upcoming shifts ({myShifts.length})
        </div>
        {myShifts.length === 0 ? (
          <div style={{
            background: C.white, border: `1.5px dashed ${C.border}`, borderRadius: 14,
            padding: "20px", textAlign: "center", marginBottom: 22,
          }}>
            <div style={{ fontSize: 24, marginBottom: 6 }}>📅</div>
            <div style={{ fontSize: 13, color: C.faint }}>No shifts assigned yet — claim one below</div>
          </div>
        ) : (
          <div style={{ display: "flex", flexDirection: "column", gap: 8, marginBottom: 22 }}>
            {myShifts.map(({ key, day, shift }) => (
              <div key={key} style={{
                background: C.white, border: `2px solid ${C.greenBorder}`, borderRadius: 13,
                padding: "12px 14px", display: "flex", alignItems: "center", gap: 12,
              }}>
                <div style={{
                  width: 38, height: 38, borderRadius: 10, background: C.greenLight,
                  display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0,
                }}>
                  <CheckIcon/>
                </div>
                <div style={{ flex: 1 }}>
                  <div style={{ fontSize: 13, fontWeight: 700, color: C.ink }}>{shift.label} · {day}</div>
                  <div style={{ fontSize: 11, color: C.faint }}>{shift.time}</div>
                </div>
                <button onClick={() => handleUnclaim(key)} style={{
                  background: "none", border: `1px solid ${C.border}`, borderRadius: 8,
                  padding: "6px 11px", fontSize: 11, fontWeight: 600, color: C.muted,
                  cursor: "pointer", fontFamily: "inherit",
                }}>Release</button>
              </div>
            ))}
          </div>
        )}

        {/* Open shifts to claim */}
        <div style={{ fontSize: 10, fontWeight: 700, color: C.faint, letterSpacing: 0.8,
          textTransform: "uppercase", marginBottom: 10 }}>
          Open shifts — claim one ({openShifts.length})
        </div>
        <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
          {openShifts.slice(0, 8).map(({ key, day, shift }) => (
            <div key={key} style={{
              background: C.white, border: `1.5px solid ${C.border}`, borderRadius: 13,
              padding: "12px 14px", display: "flex", alignItems: "center", gap: 12,
            }}>
              <div style={{ width: 8, height: 8, borderRadius: "50%", background: shift.color, flexShrink: 0 }}/>
              <div style={{ flex: 1 }}>
                <div style={{ fontSize: 13, fontWeight: 700, color: C.ink }}>{shift.label} · {day}</div>
                <div style={{ fontSize: 11, color: C.faint }}>{shift.time} · Unassigned</div>
              </div>
              <button onClick={() => handleClaim(key)} style={{
                background: `linear-gradient(135deg,${C.green},${C.greenDark})`,
                border: "none", borderRadius: 9, padding: "7px 14px",
                fontSize: 12, fontWeight: 700, color: "white", cursor: "pointer", fontFamily: "inherit",
              }}>Claim</button>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
